```cpp
const int trigPin = 10;   // Trigger pin : PWM out
const int echoPin = 7;    // Echo pin : Interrupt in

unsigned long duration;
float distance;

void setup() {
  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);
  
  Serial.begin(9600);
}

void loop() {
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);
  delayMicroseconds(10);

  duration = pulseIn(echoPin, HIGH);
  distance = (float)duration / 58.0;
  
  Serial.printf("duration : %d Distance : ", duration);
  Serial.print(distance);
  Serial.println(" [cm]");
  delay(500);
}
```

```cpp
#include "ecICAP2.h"

#include <math.h>

#include "ecGPIO2.h"
#include "ecTIM2.h"


void ICAP_init(PinName_t pinName) {
    // -------------------- 0. Map logical pin to GPIO port and timer channel --------------------
    // ecPinmap() decodes PinName_t into:
    //   - port : GPIOA / GPIOB / GPIOC ...
    //   - pin  : bit index (0–15) inside that GPIO port
    GPIO_TypeDef *port;
    unsigned int pin;
    ecPinmap(pinName, &port, &pin);

    // TIMx : pointer to timer instance (TIM1, TIM2, TIM3, TIM4, TIM5, TIM9, TIM10, TIM11)
    // TIn  : timer channel number (1–4) for this pin (TI1, TI2, TI3, TI4)
    TIM_TypeDef *TIMx;
    int TIn;
    ICAP_pinmap(pinName, &TIMx, &TIn);

    // ICn : selected input capture channel index (1–4), initially equal to mapped channel
    int ICn = TIn;

    // -------------------- GPIO configuration for alternate function (timer input) --------------------
    // 1. Configure pin as Alternate Function mode to connect it to a timer input (TIx).
    //    MODER = 10b for this pin.
    GPIO_init(pinName, AF);         // AF = 2 → Alternate Function mode

    // 2. Set output speed to high for cleaner edge timing on input (OSPEEDR = 11b).
    GPIO_ospeed(pinName, EC_HIGH);  // EC_HIGH = 3 → very high speed

    // 3. Configure GPIO AFR to select the proper alternate function for this timer:
    //    - TIM1 / TIM2 use AF1 (value 0x01).
    //    - TIM3 / TIM4 / TIM5 use AF2 (value 0x02).
    //    - TIM9 / TIM10 / TIM11 use AF3 (value 0x03).
    //    AFR[0] handles pins 0–7, AFR[1] handles pins 8–15.
    if (TIMx == TIM1 || TIMx == TIM2)
        port->AFR[pin >> 3] |= 0x01 << (4 * (pin % 8));  // AF1 for TIM1–TIM2
    else if (TIMx == TIM3 || TIMx == TIM4 || TIMx == TIM5)
        port->AFR[pin >> 3] |= 0x02 << (4 * (pin % 8));  // AF2 for TIM3–TIM5
    else if (TIMx == TIM9 || TIMx == TIM10 || TIMx == TIM11)
        port->AFR[pin >> 3] |= 0x03 << (4 * (pin % 8));  // AF3 for TIM9–TIM11

    // -------------------- TIMER base configuration --------------------
    // 1. Initialize timer update interrupt (UIF) with a default period of 1 ms.
    //    TIM_UI_init():
    //      - enables timer peripheral clock in RCC
    //      - sets PSC and ARR for the chosen base period
    //      - enables UIE in DIER
    //      - enables NVIC IRQ for this timer.
    TIM_UI_init(TIMx, 1);

    // 2. Reconfigure timer for input-capture timing:
    //    PSC and ARR values below create a 1 MHz counter clock when system clock is 84 MHz:
    //      PSC = 84 - 1 → CK_CNT = 84 MHz / 84 = 1 MHz (1.00 µs per tick)
    //      ARR = 0xFFFF → counter counts up to 65535 then generates an update event.
    TIMx->PSC = 84 - 1;       // prescaler for 1.00 µs resolution at 84.00 MHz PLL
    TIMx->ARR = 0xFFFF;       // maximum auto-reload value for longest measurable interval

    // 3. Temporarily disable the counter during capture configuration (CR1.CEN = 0).
    TIMx->CR1 &= ~TIM_CR1_CEN;

    // -------------------- Input Capture configuration (default mapping TIx → ICx) --------------------
    // 1. Connect each capture channel IC1–IC4 to corresponding timer inputs TI1–TI4:
    //    CCMR1/2 CCxS bits:
    //      CCxS = 01b → CCx channel is configured as input, mapped to TIx.
    //    All channels are prepared as input captures by default.
    TIMx->CCMR1 &= ~TIM_CCMR1_CC1S;       // clear CC1S bits (channel 1 selection)
    TIMx->CCMR1 &= ~TIM_CCMR1_CC2S;       // clear CC2S bits (channel 2 selection)
    TIMx->CCMR2 &= ~TIM_CCMR2_CC3S;       // clear CC3S bits (channel 3 selection)
    TIMx->CCMR2 &= ~TIM_CCMR2_CC4S;       // clear CC4S bits (channel 4 selection)

    TIMx->CCMR1 |= TIM_CCMR1_CC1S_0;      // CC1S = 01: TI1 → IC1
    TIMx->CCMR1 |= TIM_CCMR1_CC2S_0;      // CC2S = 01: TI2 → IC2
    TIMx->CCMR2 |= TIM_CCMR2_CC3S_0;      // CC3S = 01: TI3 → IC3
    TIMx->CCMR2 |= TIM_CCMR2_CC4S_0;      // CC4S = 01: TI4 → IC4

    // 2. Optional digital filter for input capture (commented out):
    //    These lines would adjust the sampling filter width (ICxF bits) to suppress noise.
    //    Currently disabled to keep input latency minimal.

    // 3. Input capture prescaler is not configured here (default: capture each edge).

    // 4. Configure default active edge for this ICn (rising edge).
    //    CCER bits 4n+1 (CCxP) and 4n+3 (CCxNP) control polarity:
    //      00b → capture on rising edge
    //      01b → capture on falling edge
    //      11b → capture on both edges
    //    Mask 0b1010 clears both polarity bits for the chosen channel ICn.
    TIMx->CCER &= ~(0b1010 << (4 * (ICn - 1)));  // default: rising edge for this IC channel

    // 5. Enable capture for the selected channel:
    //    CCER bit 4n (CCxE) = 1 → capture enabled on that channel.
    TIMx->CCER |= 1 << (4 * (ICn - 1));          // enable capture for ICn

    // 6. Enable interrupts for:
    //    - CCxIE: capture/compare interrupt for this ICn (DIER bit ICn).
    //    - UIE: update interrupt on overflow (DIER bit 0).
    TIMx->DIER |= 1 << ICn;       // enable CCn interrupt
    TIMx->DIER |= TIM_DIER_UIE;   // enable update interrupt

    // 7. Start timer counter (CR1.CEN = 1).
    TIMx->CR1 |= TIM_CR1_CEN;     // counter enable
}


// Configure selecting TIx–ICy mapping and edge type for a given IC channel.
void ICAP_setup(PinName_t pinName, int ICn, int edge_type) {
    // -------------------- 0. Map pin to timer and channel --------------------
    GPIO_TypeDef *port;
    unsigned int pin;
    ecPinmap(pinName, &port, &pin);

    TIM_TypeDef *TIMx;
    int CHn;                        // actual physical channel (1–4) mapped to this pin
    ICAP_pinmap(pinName, &TIMx, &CHn);

    // -------------------- 1. Disable capture and interrupt for ICn during reconfiguration --------------------
    // CCER bit 4(ICn-1) : CCxE capture enable
    TIMx->CCER &= ~(1 << (4 * (ICn - 1)));  // disable capture on channel ICn
    // DIER bit ICn : CCxIE interrupt enable
    TIMx->DIER &= ~(1 << ICn);              // disable interrupt for channel ICn

    // -------------------- 2. Configure ICn to be fed from CHn (selected TIx) --------------------
    // If ICn equals CHn, the IC channel is mapped to its “own” TIx.
    // If not, the alternate mapping (TIy) is used by setting CCxS = 10b.
    switch (ICn) {
        case 1:
            TIMx->CCMR1 &= ~TIM_CCMR1_CC1S;          // reset CC1S
            if (ICn == CHn)
                TIMx->CCMR1 |= TIM_CCMR1_CC1S_0;     // CC1S = 01: TI1 → IC1
            else
                TIMx->CCMR1 |= TIM_CCMR1_CC1S_1;     // CC1S = 10: TI2 → IC1
            break;

        case 2:
            TIMx->CCMR1 &= ~TIM_CCMR1_CC2S;          // reset CC2S
            if (ICn == CHn)
                TIMx->CCMR1 |= TIM_CCMR1_CC2S_0;     // CC2S = 01: TI2 → IC2
            else
                TIMx->CCMR1 |= TIM_CCMR1_CC2S_1;     // CC2S = 10: TI1 → IC2
            break;

        case 3:
            TIMx->CCMR2 &= ~TIM_CCMR2_CC3S;          // reset CC3S
            if (ICn == CHn)
                TIMx->CCMR2 |= TIM_CCMR2_CC3S_0;     // CC3S = 01: TI3 → IC3
            else
                TIMx->CCMR2 |= TIM_CCMR2_CC3S_1;     // CC3S = 10: TI4 → IC3
            break;

        case 4:
            TIMx->CCMR2 &= ~TIM_CCMR2_CC4S;          // reset CC4S
            if (ICn == CHn)
                TIMx->CCMR2 |= TIM_CCMR2_CC4S_0;     // CC4S = 01: TI4 → IC4
            else
                TIMx->CCMR2 |= TIM_CCMR2_CC4S_1;     // CC4S = 10: TI3 → IC4
            break;

        default:
            break;
    }

    // -------------------- 3. Configure activation edge (rising / falling / both) --------------------
    // CCER polarity bits per channel:
    //   CCxP  (bit 4n+1) and CCxNP (bit 4n+3):
    //      00b → capture on rising edge
    //      01b → capture on falling edge
    //      11b → capture on both edges
    TIMx->CCER &= ~(0b1010 << (4 * (ICn - 1)));  // clear CCxP / CCxNP bits

    switch (edge_type) {
        case IC_RISE:                                      // rising edge only
            // CCxP = 0, CCxNP = 0 already set by mask
            break;

        case IC_FALL:                                      // falling edge only
            TIMx->CCER |= 0b0010 << (4 * (ICn - 1));       // CCxP = 1, CCxNP = 0
            break;

        case IC_BOTH:                                      // both edges
            TIMx->CCER |= 0b1010 << (4 * (ICn - 1));       // CCxP = 1, CCxNP = 1
            break;
    }

    // -------------------- 4. Re-enable capture and interrupt for ICn --------------------
    TIMx->CCER |= 1 << (4 * (ICn - 1));   // enable capture on ICn
    TIMx->DIER |= 1 << ICn;              // enable CCn interrupt
}


// Configure timer counter step size in microseconds for the pin's timer.
void ICAP_counter_us(PinName_t pinName, int usec) {
    // -------------------- 0. Map pin to timer --------------------
    GPIO_TypeDef *port;
    unsigned int pin;
    ecPinmap(pinName, &port, &pin);

    TIM_TypeDef *TIMx;
    int CHn;
    ICAP_pinmap(pinName, &TIMx, &CHn);

    // -------------------- Timer prescaler and ARR configuration --------------------
    // System clock assumed 84.00 MHz (PLL).
    // PSC = 84 * usec - 1:
    //   CK_CNT = 84 MHz / (PSC + 1) = 84 MHz / (84 * usec) = (1 / usec) MHz
    //   → one counter tick = usec microseconds.
    // Example: usec = 10 → CK_CNT = 0.10 MHz → 10.00 µs per tick.
    TIMx->PSC = 84 * usec - 1;

    // ARR at maximum (0xFFFF) allows longest possible measurement interval
    // before counter overflow.
    TIMx->ARR = 0xFFFF;
}


// Read captured value from the selected input capture channel ICn.
uint32_t ICAP_capture(TIM_TypeDef *TIMx, uint32_t ICn) {
    uint32_t capture_Value;

    // CCRx register holds the timer count latched at the configured input edge.
    if (ICn == 1)
        capture_Value = TIMx->CCR1;
    else if (ICn == 2)
        capture_Value = TIMx->CCR2;
    else if (ICn == 3)
        capture_Value = TIMx->CCR3;
    else
        capture_Value = TIMx->CCR4;

    return capture_Value;
}


// Check Capture/Compare interrupt flag (CCxIF) for a given channel.
uint32_t is_CCIF(TIM_TypeDef *TIMx, uint32_t ccNum) {
    // SR bit ccNum (1–4):
    //   bit1 = CC1IF, bit2 = CC2IF, bit3 = CC3IF, bit4 = CC4IF.
    return (TIMx->SR & (0x1UL << ccNum)) != 0;
}


// Clear Capture/Compare interrupt flag for a given channel.
void clear_CCIF(TIM_TypeDef *TIMx, uint32_t ccNum) {
    // CCxIF is cleared by writing 0 to the corresponding bit in SR.
    TIMx->SR &= ~(1 << ccNum);
}


// Map PinName_t to corresponding timer instance and channel index used for Input Capture.
void ICAP_pinmap(PinName_t pinName, TIM_TypeDef **TIMx, int *chN) {
    GPIO_TypeDef *port;
    unsigned int pin;
    ecPinmap(pinName, &port, &pin);

    // -------------------- Port A pin mapping --------------------
    if (port == GPIOA) {
        switch (pin) {
            case 0:  *TIMx = TIM2; *chN = 1; break;   // PA0 → TIM2_CH1
            case 1:  *TIMx = TIM2; *chN = 2; break;   // PA1 → TIM2_CH2
            case 5:  *TIMx = TIM2; *chN = 1; break;   // PA5 → TIM2_CH1 (alternate)
            case 6:  *TIMx = TIM3; *chN = 1; break;   // PA6 → TIM3_CH1
            // case 7: *TIMx = TIM1; *chN = 1N; break; // reserved / commented
            case 8:  *TIMx = TIM1; *chN = 1; break;   // PA8 → TIM1_CH1
            case 9:  *TIMx = TIM1; *chN = 2; break;   // PA9 → TIM1_CH2
            case 10: *TIMx = TIM1; *chN = 3; break;   // PA10 → TIM1_CH3
            case 15: *TIMx = TIM2; *chN = 1; break;   // PA15 → TIM2_CH1
            default: break;
        }
    }
    // -------------------- Port B pin mapping --------------------
    else if (port == GPIOB) {
        switch (pin) {
            // case 0: *TIMx = TIM1; *chN = 2N; break; // reserved / commented
            // case 1: *TIMx = TIM1; *chN = 3N; break; // reserved / commented
            case 3:  *TIMx = TIM2; *chN = 2; break;   // PB3 → TIM2_CH2
            case 4:  *TIMx = TIM3; *chN = 1; break;   // PB4 → TIM3_CH1
            case 5:  *TIMx = TIM3; *chN = 2; break;   // PB5 → TIM3_CH2
            case 6:  *TIMx = TIM4; *chN = 1; break;   // PB6 → TIM4_CH1
            case 7:  *TIMx = TIM4; *chN = 2; break;   // PB7 → TIM4_CH2
            case 8:  *TIMx = TIM4; *chN = 3; break;   // PB8 → TIM4_CH3
            case 9:  *TIMx = TIM4; *chN = 4; break;   // PB9 → TIM4_CH4
            case 10: *TIMx = TIM2; *chN = 3; break;   // PB10 → TIM2_CH3
            default: break;
        }
    }
    // -------------------- Port C pin mapping --------------------
    else if (port == GPIOC) {
        switch (pin) {
            case 6:  *TIMx = TIM3; *chN = 1; break;   // PC6 → TIM3_CH1
            case 7:  *TIMx = TIM3; *chN = 2; break;   // PC7 → TIM3_CH2
            case 8:  *TIMx = TIM3; *chN = 3; break;   // PC8 → TIM3_CH3
            case 9:  *TIMx = TIM3; *chN = 4; break;   // PC9 → TIM3_CH4
            default: break;
        }
    }
}
```

## Main Echo FSM (Rising / Falling Edge)

### Definitions
- **States**
  - **S0**: Idle, waiting for rising edge (no active echo pulse)
  - **S1**: Echo pulse active, waiting for falling edge (pulse width measurement in progress)
  1 **S2**: Measurement completed, `timeInterval` updated

- **Input Events**
  - **ev = IC1**: Rising edge detected (`CC1IF = 1`, TIM4_CH1)
  - **ev = IC2**: Falling edge detected (`CC2IF = 1`, TIM4_CH2)
  - **ev = none**: No capture event

### FSM Table

| **State** | **next[ev = none]** | **next[ev = IC1]** | **next[ev = IC2]** | **Out [ovf_cnt, time1, time2, timeInterval]** |
|:--------:|:-------------------:|:------------------:|:------------------:|:----------------------------------------------|
| **S0**   | S0                  | S1                 | S0                 | on IC1: `time1 ← CCR1`, `ovf_cnt ← 0`          |
| **S1**   | S1                  | S1                 | S2                 | on IC2: `time2 ← CCR2`, `timeInterval ← ((time2 − time1) + ovf_cnt·(ARR+1)) / 100`, `ovf_cnt ← 0` |
| **S2**   | S0                  | S1                 | S2                 | measurement complete; S0 on next cycle         |

### Discussion

1. There can be an over-capture case, when a new capture interrupt occurs before reading the CCR value. When does it occur and how can you calculate the time span accurately between two captures?

	An over-capture happens when a new input-capture edge occurs before the software has read the previous CCR value.  
	When this occurs, the timer overwrites the older captured timestamp and raises the “over-capture” flag (CCxOF).  
	Because the previous capture value is lost, the exact time difference between the two consecutive edges cannot be recovered.

	Over-capture occurs when:
	- The input signal edges arrive too quickly (high frequency).
	- The interrupt service routine is delayed or blocked.
	- Other interrupts have higher priority and increase latency.
	- The timer tick resolution is very small, causing captures to happen too often.
	- A rising or falling edge arrives again before CCxIF is cleared.

	When over-capture does **not** occur, the time difference between two edges is computed using:

	- `CCR2 - CCR1` → raw difference between captured timestamps  
	- `ovf_cnt` → number of timer overflows during the echo pulse  
	- `ARR + 1` → size of one full counter cycle  
	- `T_tick` → duration of one timer tick  

	The total tick count is:
	ΔN = (CCR2 - CCR1) + ovf_cnt * (ARR + 1)
	The time duration is:
	Δt = ΔN * T_tick
	This method provides an accurate time span as long as no over-capture occurs.
	
2. In the tutorial, what is the accuracy when measuring the period of 1Hz square wave? Show your result.

	In the tutorial setup, the timer is configured so that each timer tick represents exactly **10 microseconds**.

	This comes from:
	- System clock = 84 MHz  
	- Prescaler = 839  
	- Timer frequency = 100 kHz  
	- Timer tick = 1 / 100 kHz = **10 microseconds**

	A 1 Hz square wave has a true period of **1 second = 1,000,000 microseconds**.

	Because Input Capture timestamps are quantized to one timer tick, the best achievable measurement resolution is:
	± 1 timer tick = ± 10 microseconds
	The relative accuracy is:
	Accuracy = (10 microseconds / 1,000,000 microseconds) * 100%  
	Accuracy = 0.001%

	### **Final Result**
	- Timer tick resolution: **10 microseconds**
	- Maximum error: **±10 microseconds**
	- Relative accuracy: **0.001%**

```c
#include "math.h"

// -------------------- Global timing and distance variables --------------------

// Counts the number of TIM4 update events (overflow of CNT from ARR → 0)
uint32_t ovf_cnt = 0;

// Stores the final calculated distance in centimeters
float distance = 0;

// Stores the measured echo pulse width (round-trip time of ultrasonic wave) in milliseconds
float timeInterval = 0;

// Timestamp captured at the rising edge of the ECHO signal (start of echo pulse)
float time1 = 0;

// Timestamp captured at the falling edge of the ECHO signal (end of echo pulse)
float time2 = 0;

// -------------------- Pin mapping --------------------

// Ultrasonic TRIG pin: PA_6 is configured as PWM output (TIM3_CH1 by EC_HAL pinmap)
#define TRIG PA_6

// Ultrasonic ECHO pin: PB_6 is configured as TIM4 input capture (TIM4_CH1/CH2)
#define ECHO PB_6


void setup(void);


int main(void){
    // Local variable (not used in current logic, left as-is)
    int count_test = 0;

    // System and peripheral configuration
    setup();

    // Main loop: periodically prints the measured distance
    while(1){
        // --------------------------------------------------------------------
        // Distance computation from measured time interval
        // timeInterval: echo pulse duration [ms]
        // Speed of sound ≈ 340.00 m/s = 340.00 mm/ms
        // distance[mm] = timeInterval[ms] * 340.00 [mm/ms] / 2.00  (round-trip → one-way)
        // distance[cm] = distance[mm] / 10.00
        // --------------------------------------------------------------------
        distance = (float)timeInterval * 340.0f / 2.0f / 10.0f;

        // --------------------------------------------------------------------
        // Manual formatting to print distance with exactly two decimal places
        // avoids float printing artifacts and keeps output stable as "xxx.yy cm".
        // --------------------------------------------------------------------
        int dist100  = (int)(distance * 100.0f);  // distance × 100.00 → integer in centi-centimeters
        int dist_int = dist100 / 100;             // integer part of distance
        int dist_frac = dist100 % 100;            // fractional part (0~99)

        // Handles negative numbers if any numerical underflow/rounding occurs
        if(dist_frac < 0) dist_frac = -dist_frac;

        // Prints formatted distance as "<integer>.<two-digit fractional> cm"
        printf("%d.%02d cm\r\n", dist_int, dist_frac);

        // Waits 1000.00 ms between prints using SysTick-based delay
        delay_ms(1000);
    }
}


// -----------------------------------------------------------------------------
// TIM4 global interrupt service routine
// - Handles both update events (overflow) and input capture events on ECHO.
// - Uses EC_HAL helper functions operating on TIM4 registers.
// -----------------------------------------------------------------------------
void TIM4_IRQHandler(void){
    // -------------------- Update Event (Overflow) Handling --------------------
    // is_UIF(TIM4) checks TIM4->SR bit 0 (UIF: Update Interrupt Flag).
    if(is_UIF(TIM4)){
        // Increments overflow count whenever CNT rolls from ARR to 0.
        ovf_cnt++;

        // Clears UIF flag by writing 0 to bit 0 of TIM4->SR.
        clear_UIF(TIM4);
    }

    // -------------------- Capture/Compare Channel 1: Rising Edge --------------------
    // is_CCIF(TIM4, 1) checks TIM4->SR bit 1 (CC1IF: Capture/Compare 1 Interrupt Flag).
    if(is_CCIF(TIM4, 1)){
        // Captures the current CNT value at rising edge into CCR1 and returns it.
        // time1 represents the starting tick count of the echo pulse.
        time1 = ICAP_capture(TIM4, 1);

        // Resets overflow counter at the beginning of a new echo pulse.
        // From this rising edge until the corresponding falling edge,
        // ovf_cnt counts how many times CNT overflowed.
        ovf_cnt = 0;

        // Clears CC1IF flag in TIM4->SR for channel 1.
        clear_CCIF(TIM4, 1);
    }

    // -------------------- Capture/Compare Channel 2: Falling Edge --------------------
    // is_CCIF(TIM4, 2) checks TIM4->SR bit 2 (CC2IF: Capture/Compare 2 Interrupt Flag).
    else if(is_CCIF(TIM4, 2)){
        // Captures the current CNT value at falling edge into CCR2 and returns it.
        // time2 represents the ending tick count of the echo pulse.
        time2 = ICAP_capture(TIM4, 2);

        // ----------------------------------------------------------------
        // Time interval reconstruction considering possible CNT overflow
        //
        // ICAP_counter_us(ECHO, 10) configured TIM4 such that:
        //   - PSC and ARR are chosen so that one CNT increment = 10.00 µs.
        //   - CNT counts up from 0 to ARR, then overflows.
        //
        // (time2 - time1): difference in timer ticks within current 16-bit range.
        // ovf_cnt * (TIM4->ARR + 1): additional ticks from full overflows.
        //
        // Total ticks = (time2 - time1) + ovf_cnt * (ARR + 1)
        //
        // Each tick = 10.00 µs = 0.01 ms
        // Therefore:
        //   timeInterval[ms] = total_ticks * 0.01 = total_ticks / 100.00
        // ----------------------------------------------------------------
        timeInterval = ((time2 - time1) + ovf_cnt * ((TIM4->ARR) + 1)) / 100.0f;

        // Overflow counter is reset to prepare for the next echo measurement.
        ovf_cnt = 0;

        // Clears CC2IF flag in TIM4->SR for channel 2.
        clear_CCIF(TIM4, 2);
    }
}


// -----------------------------------------------------------------------------
// System and peripheral initialization
// - Configures system clock, SysTick, UART, PWM trigger for ultrasonic TRIG,
//   and TIM4 input capture for ultrasonic ECHO.
// -----------------------------------------------------------------------------
void setup(){

    // -------------------- System Clock & SysTick --------------------
    // Configures PLL to generate 84.00 MHz system clock from internal HSI.
    // Internally sets RCC->PLLCFGR, RCC->CFGR, FLASH->ACR.
    RCC_PLL_init();

    // Initializes SysTick to generate interrupts every 1.00 ms.
    // SysTick->LOAD = 84,000,000 / 1000 - 1.
    // SysTick_Handler increments msTicks used by delay_ms().
    SysTick_init();

    // -------------------- UART2 for printf output --------------------
    // Initializes USART2 on PA_2 (TX), PA_3 (RX) with default baud rate (9600).
    // Internally sets GPIOA->MODER/AFR and USART2->BRR/CR1.
    UART2_init();

    // -------------------- TRIG pin basic GPIO configuration --------------------
    // Configures TRIG (PA_6) output type as push-pull (OTYPER bit = 0).
    GPIO_otype(TRIG, 0);

    // Disables pull-up and pull-down resistors on TRIG (PUPDR = 00).
    GPIO_pupd(TRIG, 0);

    // Sets TRIG speed to fast (OSPEEDR = 10).
    GPIO_ospeed(TRIG, EC_FAST);

    // -------------------- PWM configuration for TRIG (ultrasonic trigger) --------------------
    // PWM_init(TRIG):
    //   - Maps PA_6 → TIM3_CH1 (by PWM_pinmap).
    //   - Configures PA_6 as AF, sets AFR for TIM3.
    //   - Initializes TIM3 with 1.00 ms default period and enables PWM Mode 1.
    PWM_init(TRIG);

    // Sets PWM period to 50,000.00 µs = 50.00 ms.
    // TIM3->PSC and TIM3->ARR are adjusted to achieve this period.
    PWM_period_us(TRIG, 50000);

    // Sets PWM high pulse width to 10.00 µs.
    // Corresponds to trigger pulse width for the ultrasonic sensor.
    PWM_pulsewidth_us(TRIG, 10);

    // Result:
    //   TRIG (PA_6) outputs a 10.00 µs HIGH pulse every 50.00 ms automatically.

    // -------------------- Input Capture configuration for ECHO --------------------
    // ICAP_init(ECHO):
    //   - Maps PB_6 → TIM4 input capture channels (TIM4_CH1/CH2).
    //   - Configures PB_6 as AF with correct AFR setting for TIM4.
    //   - Initializes TIM4, enables UIE and CCx interrupts.
    ICAP_init(ECHO);

    // Disables pull-up and pull-down resistors on ECHO pin.
    GPIO_pupd(ECHO, 0);

    // Configures TIM4 prescaler so that one CNT tick = 10.00 µs.
    //   PSC = 84 * usec - 1  → for usec=10, PSC = 839.
    //   ARR = 0xFFFF (max count before overflow).
    ICAP_counter_us(ECHO, 10);

    // Sets TIM4_IC1 to capture on rising edge for ECHO.
    // ICAP_setup(ECHO, 1, IC_RISE):
    //   - Connects proper TIx to IC1.
    //   - Configures CC1S bits and CC1P/CC1NP for rising edge detection.
    ICAP_setup(ECHO, 1, IC_RISE);

    // Sets TIM4_IC2 to capture on falling edge for ECHO.
    // ICAP_setup(ECHO, 2, IC_FALL):
    //   - Connects proper TIx to IC2.
    //   - Configures CC2S bits and CC2P/CC2NP for falling edge detection.
    ICAP_setup(ECHO, 2, IC_FALL);

    // After this setup:
    //   - TRIG generates periodic trigger pulses via TIM3 PWM on PA_6.
    //   - ECHO pulse is measured by TIM4 input capture:
    //        Rising edge → CCR1 (time1),
    //        Falling edge → CCR2 (time2),
    //        Overflows counted in ovf_cnt.
    //   - TIM4_IRQHandler reconstructs pulse width in milliseconds and
    //     stores it in timeInterval for distance calculation in the main loop.
}
```
